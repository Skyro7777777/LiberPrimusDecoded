    99→def hex_pairs_to_bytes(hex_str: str) -> bytes:
   100→    """Convert a hex string to bytes (must be even length)."""
   101→    if len(hex_str) % 2: hex_str = "0" + hex_str
   102→    return bytes.fromhex(hex_str)
   103→
   104→
   105→def bytes_to_rune_key(b: bytes, mod: int = MOD) -> str:
   106→    """Convert bytes to a rune-key (each byte mod 29 → rune)."""
   107→    return "".join(DEC_TO_RUNE[byte % mod] for byte in b)
   108→
   109→
   110→def hex_digits_to_rune_key(hex_str: str, mod: int = MOD) -> str:
   111→    """Convert each hex digit (0-15) to a rune (mod 29 = identity)."""
   112→    return "".join(DEC_TO_RUNE[int(c, 16) % mod] for c in hex_str)
   113→
   114→
   115→def dec_digits_to_rune_key(dec_str: str, mod: int = MOD) -> str:
   116→    """Convert each decimal digit (0-9) to a rune (mod 29 = identity)."""
   117→    return "".join(DEC_TO_RUNE[int(c) % mod] for c in dec_str if c.isdigit())
   118→
   119→
   120→def score_pt(pt_runes: str) -> Tuple[float, str]:
   121→    """Score a plaintext (in runes) — convert to Latin then english_score."""
   122→    latin = runes_to_latin(pt_runes)
   123→    score = english_score(latin)
   124→    return score, latin

   163→    }

   267→# ATTACK 2 — HILL-CLIMBING AUTOKEY PRIMER DISCOVERY
   268→# ============================================================================
   269→
   270→def hill_climb_autokey(L: int, mode: str, cipher_runes: str, n_iter: int = 5000,
   271→                       t_start: float = 2.0, t_end: float = 0.1, rng=None,
   272→                       verbose: bool = False) -> Dict:
   273→    """
   274→    Simulated-annealing hill-climb on an autokey primer of length L.
   275→
   276→    Decrypt first len(cipher_runes) runes with autokey_vigenere(mode).
   277→    Score with english_score on Latin output.
   278→    Mutate 1 rune in primer; accept if better, or worse with prob exp((new-old)/T).
   279→    """
   280→    if rng is None: rng = random.Random()
   281→    n = len(cipher_runes)
   282→    # Random initial primer
   283→    primer_decs = [rng.randint(0, MOD-1) for _ in range(L)]
   284→    primer_runes = "".join(DEC_TO_RUNE[d] for d in primer_decs)
   285→    pt = autokey_vigenere(cipher_runes, primer_runes, mode=mode, decrypt=True)
   286→    cur_score = english_score(runes_to_latin(pt))
   287→    best_score = cur_score
   288→    best_primer = primer_decs[:]
   289→    best_pt = pt
   290→
   291→    # Temperature schedule: linear in 1/T for smoother cooling
   292→    inv_t_start = 1.0 / t_start
   293→    inv_t_end = 1.0 / t_end
   294→    for it in range(n_iter):
   295→        # Linear interpolation in inverse-T (geometric cooling equivalent)
   296→        frac = it / max(1, n_iter - 1)
   297→        inv_t = inv_t_start + frac * (inv_t_end - inv_t_start)
   298→        T = 1.0 / inv_t
   299→
   300→        # Mutate 1 random position to a random new rune
   301→        pos = rng.randint(0, L-1)
   302→        old_val = primer_decs[pos]
   303→        new_val = rng.randint(0, MOD-1)
   304→        if new_val == old_val:
   305→            new_val = (new_val + 1) % MOD
   306→        primer_decs[pos] = new_val
   307→        primer_runes = "".join(DEC_TO_RUNE[d] for d in primer_decs)
   308→        pt = autokey_vigenere(cipher_runes, primer_runes, mode=mode, decrypt=True)
   309→        new_score = english_score(runes_to_latin(pt))
   310→
   311→        delta = new_score - cur_score
   312→        if delta >= 0 or rng.random() < pow(2.718281828459045, delta / max(T, 1e-6)):
   313→            # Accept
   314→            cur_score = new_score
   315→            if new_score > best_score:
   316→                best_score = new_score
   317→                best_primer = primer_decs[:]
   318→                best_pt = pt
   319→        else:
   320→            # Reject — revert
   321→            primer_decs[pos] = old_val
   322→
   323→    best_primer_runes = "".join(DEC_TO_RUNE[d] for d in best_primer)
   324→    best_latin = runes_to_latin(best_pt)
   325→    if verbose:
   326→        print(f"  L={L:3d} mode={mode:9s} best_score={best_score:.3f} primer={best_primer_runes[:30]}")
   327→        print(f"    pt[:80]: {best_latin[:80]}")
   328→    return {
   329→        "L": L, "mode": mode, "best_score": round(best_score, 3),
   330→        "best_primer_runes": best_primer_runes,
   331→        "best_primer_latin": decimals_to_latin(best_primer),
   332→        "best_pt_preview": best_latin[:120],
   333→    }

   398→def zeckendorf_decomp(n: int) -> List[int]:
   399→    """Return Fibonacci indices (1-indexed) in the Zeckendorf decomposition of n.
   400→    Uses Fibonacci numbers F(1)=1, F(2)=2, F(3)=3, F(4)=5, F(5)=8, ... (avoid the duplicate 1)."""
   401→    if n <= 0: return []
   402→    # Build Fibonacci list with no leading 1 (start F(2)=2 to avoid the 1,1 ambiguity)
   403→    fibs = [1, 2]
   404→    while fibs[-1] < n:
   405→        fibs.append(fibs[-1] + fibs[-2])
   406→    # Greedy from largest
   407→    indices = []
   408→    remaining = n
   409→    for i in range(len(fibs)-1, -1, -1):
   410→        if fibs[i] <= remaining:
   411→            indices.append(i+1)  # 1-indexed
   412→            remaining -= fibs[i]
   413→        if remaining == 0:
   414→            break
   415→    return indices

   487→def xor_stream_decrypt(cipher_runes: str, key_bytes: bytes) -> str:
   488→    """XOR each rune-decimal (0-28, treated as 1 byte) with the key (cycled).
   489→    Result mod 29 → rune."""
   490→    cipher_decs = runes_to_decimals(cipher_runes)
   491→    out = []
   492→    for i, cd in enumerate(cipher_decs):
   493→        kb = key_bytes[i % len(key_bytes)]
   494→        pd = (cd ^ kb) % MOD
   495→        out.append(DEC_TO_RUNE[pd])
   496→    return "".join(out)
   497→

   499→def add_stream_decrypt(cipher_runes: str, key_decs: List[int]) -> str:
   500→    """Subtract each key value from cipher-dec (mod 29) — pure Vigenère-style keystream."""
   501→    cipher_decs = runes_to_decimals(cipher_runes)
   502→    out = []
   503→    for i, cd in enumerate(cipher_decs):
   504→        kd = key_decs[i % len(key_decs)]
   505→        pd = (cd - kd) % MOD
   506→        out.append(DEC_TO_RUNE[pd])
   507→    return "".join(out)
