   323→    def crop_ct_to_keylen_with_interrupter(self, ct, keylen):
   324→        '''
   325→            Crop the CT so that it is just long enough to apply the key, with all possible
   326→            passed interrupters. This is used so that when a very long CT is passed we don't get
   327→            interrupters for the whole CT, just the part that the key can be applied to.
   328→        :param ct: ct to crop
   329→        :param keylen: number of runes in key
   330→        :return: the cropped CT
   331→        '''
   332→        # use normal gematria position for interrupters
   333→        ct_p = self.gem.encode_as_position(ct,'forward',0)
   334→        ct_p_len = len(ct_p)
   335→        ct_out = ct[:keylen]
   336→        ct_out_max_count = max([ct_out.count(x) for x in set(ct_out)])
   337→        while 1:
   338→            ''' check if keylength + max count == length of return text (then we can return) '''
   339→            ''' check if keylength + max count > length of passed ct (then we must return) '''
   340→            ''' else add to the return list '''
   341→            if keylen + ct_out_max_count == len(ct_out):
   342→                return ct_out
   343→            elif keylen + ct_out_max_count > ct_p_len:
   344→                return ct_out
   345→            else:
   346→                ct_out = ct[:keylen + ct_out_max_count]
   347→                ct_out_max_count = max([ct_out.count(x) for x in set(ct_out)])
